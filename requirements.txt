# Pinned per spec/architecture.md section 1.
# Must install in a single Colab/Kaggle cell with no compiler and no API key.
#
# Interpreters: Python 3.11 AND 3.13, both supported, both target execution
# environments - Kaggle runs 3.11, Colab runs 3.13 (spec/architecture.md
# section 1). Verified to install with no compiler invoked and pass pytest on
# 3.11.9 and on 3.13.7.
#
# Bound style: lower bounds for the ambient scientific stack, because Kaggle and
# Colab ship their own numpy/pandas/scipy and an exact pin forces a downgrade
# that breaks the one-cell install (AGENTS.md section 3). An upper bound is
# added only where there is a stated reason to expect a breaking major release.

numpy>=1.26,<3
scipy>=1.12

# pandas capped to the EXECUTION environment, not the dev box - see D-015.
# Developing on pandas 3 means writing Parquet locally that the machines doing
# the bulk generation have never been tested against, and finding out at M1
# six hours into a Kaggle job.
pandas>=2.2,<3

# pyarrow: no upper bound - see D-019, which supersedes the pyarrow half of
# D-015. The <19 cap was set from Colab's 2026-08-30 ambient version (18.1.0);
# by 2026-09-14 Colab shipped 23.0.1 and Kaggle 24.0.0, so the cap was forcing
# a downgrade on both platforms' base images and breaking datasets/bigframes
# there. Nothing here depends on pyarrow internals - rows are float64 and
# strings, and the Parquet format is stable across this range.
pyarrow>=15

networkx>=3.2

# Label-determining: a major release that changes decoding or sampling
# behaviour would silently invalidate every label already generated, so both
# are bounded and that fails at install time instead. ldpc's version enters
# protocol_hash (CONTRACT.md INV-6). stim's does not: it is stored per row as
# stim_version (D-027).
stim>=1.14,<2
ldpc>=2.1,<3

# Not used by this project and off the label path (D-026). Installed only
# because ldpc depends on it (and imports it at import time). Its version never
# entered protocol_hash; an earlier comment here said it did.
sinter>=1.14,<2

scikit-learn>=1.4
lightgbm>=4.3
pytest>=8.0

# Pulled in transitively, deliberately not pinned here:
#   matplotlib  <- sinter
#   pymatching  <- ldpc
#
# On sinter 1.15 vs 1.16: that split is WHEEL AVAILABILITY, not interpreter
# version. sinter 1.16.0 is sdist-only on PyPI - no wheel for any Python. A
# plain `pip install` builds it from the sdist (pure Python, no compiler) and
# gives 1.16.0 on 3.11 and 3.13 alike. Only a wheels-only install
# (--only-binary=:all:) falls back to 1.15.0, and it does so on every
# interpreter. See AGENT_LOG (n).
#
# NOT included, deliberately:
#   torch / torch-geometric  -> M3 only, optional, see AGENTS.md section 3
#   anything needing an API key or a paid tier -> banned by CONTRACT.md INV-9
